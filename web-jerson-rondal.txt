function addPlaneToMap(f){

        const STORAGE_KEY = "guiver_oct2026_seo_v5";

        function loadReportsFromStorage() {
            const raw = localStorage.getItem(STORAGE_KEY);
            let custom = [];
            if (raw) {
                try { custom = JSON.parse(raw); } catch(e) { custom = []; }
            }
            const map = new Map();
            custom.forEach(r => map.set(r.id, r));
            OCTOBER_REPORTS.forEach(r => {
                if (!map.has(r.id)) map.set(r.id, r);
            });
            return Array.from(map.values());
        }

        let loadedReports = loadReportsFromStorage();
        let currentCat = 'all';
        let activeReport = loadedReports[0];

        function parseUploadedHTML(htmlString, fallbackName) {
            const parser = new DOMParser();
            const doc = parser.parseFromString(htmlString, 'text/html');
            const h2s = Array.from(doc.querySelectorAll('.t-title h2, h2')).map(e => e.textContent.trim()).filter(Boolean);
            const sub = doc.querySelector('.hero .sub, .doc-sub')?.textContent.trim() || '';
            const titleTag = doc.querySelector('title')?.textContent.trim() || '';

            let title = h2s.length > 0 ? h2s.slice(0, 2).join(' | ') : (titleTag || fallbackName.replace(/\.html?$/i, '').replace(/_/g, ' '));
            let dateStr = new Date().toLocaleDateString('es-ES', { day: '2-digit', month: 'long', year: 'numeric' });
            const dm = (sub + ' ' + titleTag).match(/(\d{1,2}\s+(?:al\s+\d{1,2}\s+)?de\s+[a-záéíóú]+(?:\s+de\s+\d{4})?|\d{1,2}\s*OCT\s*\d{4})/i);
            if (dm) dateStr = dm[1];

            const tags = [];
            const low = htmlString.toLowerCase();
            if (low.includes('ormuz') || low.includes('irán')) tags.push('Ormuz');
            if (low.includes('taiwán') || low.includes('ccg')) tags.push('Taiwán');
            if (low.includes('mar rojo') || low.includes('hutí') || low.includes('riad')) tags.push('Mar Rojo');
            if (tags.length === 0) tags.push('Deep Dive');

            const isWeekly = low.includes('semanal') || low.includes('11 al 18');
            return {
                id: 'custom-' + Date.now() + '-' + Math.random().toString(36).substring(2, 6),
                date: dateStr,
                category: isWeekly ? 'semanal' : 'diario',
                badge: isWeekly ? 'DOSSIER SEMANAL' : 'REPORTE DIARIO',
                badgeClass: isWeekly ? 'weekly' : '',
                theaters: tags,
                title: title,
                html: htmlString,
                custom: true
            };
        }

        function setCategory(cat, btn) {

        }

        function renderReports() {
            const listEl = document.getElementById('reportsList');
            const q = document.getElementById('searchBox').value.toLowerCase();
            listEl.innerHTML = '';

            const filtered = loadedReports.filter(r => {
                const matchCat = currentCat === 'all' || r.category === currentCat;
                const matchQ = r.title.toLowerCase().includes(q) ||
                               r.date.toLowerCase().includes(q) ||
                               (r.theaters && r.theaters.join(' ').toLowerCase().includes(q));
                return matchCat && matchQ;
            });

            document.getElementById('totalReportsBadge').textContent = `${loadedReports.length} DOSSIERES COMPLETOS`;

            filtered.forEach(rep => {
                const div = document.createElement('div');
                const isAct = activeReport && activeReport.id === rep.id;
                div.className = 'report-card' + (isAct ? ' active' : '');
                div.onclick = () => selectReport(rep);

                const tagsHTML = (rep.theaters || []).map(t => `<span class="rc-tag">${t}</span>`).join('');
                const delHTML = rep.custom ? `<button class="rc-del" title="Eliminar reporte" onclick="deleteReport('${rep.id}', event)">🗑️</button>` : '';

                    <div class="rc-footer">
                        <div class="rc-tags">${tagsHTML}</div>
                        ${delHTML}
                    </div>

                listEl.appendChild(div);
            });
        }

        function deleteReport(id, ev) {
            ev.stopPropagation();
            loadedReports = loadedReports.filter(r => r.id !== id);
            localStorage.setItem(STORAGE_KEY, JSON.stringify(loadedReports.filter(r => r.custom)));
            if (activeReport && activeReport.id === id) activeReport = loadedReports[0];
            selectReport(activeReport);
        }

        async function handleFiles(e) {
            const files = Array.from(e.target.files);
            for (const file of files) {
                const text = await file.text();
                const meta = parseUploadedHTML(text, file.name);
                loadedReports.unshift(meta);
                activeReport = meta;
            }
            localStorage.setItem(STORAGE_KEY, JSON.stringify(loadedReports.filter(r => r.custom)));
            toggleUploadModal(false);
            selectReport(activeReport);
        }

        function savePastedReport() {
            const code = document.getElementById('pasteCode').value.trim();
            const customTitle = document.getElementById('pasteTitle').value.trim();
            if (!code) return alert("Pega el código HTML del reporte diario.");
            const meta = parseUploadedHTML(code, customTitle || "Reporte Diario GUIVER");
            if (customTitle) meta.title = customTitle;
            loadedReports.unshift(meta);
            localStorage.setItem(STORAGE_KEY, JSON.stringify(loadedReports.filter(r => r.custom)));
            document.getElementById('pasteCode').value = '';
            document.getElementById('pasteTitle').value = '';
            togglePasteModal(false);
            selectReport(meta);
        }

        async function triggerFolderSync() {
            if (!window.showDirectoryPicker) return toggleUploadModal(true);
            try {
                const dir = await window.showDirectoryPicker();
                let count = 0;
                for await (const entry of dir.values()) {
                    if (entry.kind === 'file' && (entry.name.endsWith('.html') || entry.name.endsWith('.htm'))) {
                        const file = await entry.getFile();
                        const text = await file.text();
                        const meta = parseUploadedHTML(text, entry.name);
                        loadedReports.unshift(meta);
                        activeReport = meta;
                        count++;
                    }
                }
                localStorage.setItem(STORAGE_KEY, JSON.stringify(loadedReports.filter(r => r.custom)));
                if (activeReport) selectReport(activeReport);
                alert(`Se han sincronizado ${count} reportes HTML.`);
            } catch (err) {}
        }

            try {
                const doc = document.getElementById('reportFrame').contentDocument;
                if (!doc) return;
                const root = doc.documentElement;
                root.setAttribute('data-theme', root.getAttribute('data-theme') === 'dark' ? 'light' : 'dark');
            } catch (e) {}
        }

        function downloadCurrentReport() {
            if (!activeReport) return;
            const blob = new Blob([activeReport.html], { type: 'text/html;charset=utf-8' });
            const a = document.createElement('a');
            a.href = URL.createObjectURL(blob);
            a.download = `GUIVER_${activeReport.date.replace(/\s+/g, '_')}.html`;
            a.click();
        }

        function openInNewTab() {
            if (!activeReport) return;
            const blob = new Blob([activeReport.html], { type: 'text/html;charset=utf-8' });
            window.open(URL.createObjectURL(blob), '_blank');
        }

        function printReport() {
            const frame = document.getElementById('reportFrame');
            frame.contentWindow.focus();
            frame.contentWindow.print();
        }

        function toggleFullscreen() {
            const frame = document.getElementById('reportFrame');
            if (frame.requestFullscreen) frame.requestFullscreen();
        }

        selectReport(loadedReports[0]);
    </script>
